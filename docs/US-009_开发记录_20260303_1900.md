### 3. 绑定模式切换

**文件**: `src/App.tsx`

新增绑定模式切换功能：

- **逐个绑定模式**：每个场景单独绑定角色参考图
- **全局绑定模式**：所有场景共享角色绑定

```tsx
<Radio.Group
  value={bindMode}
  onChange={e => {
    setBindMode(e.target.value);
    if (e.target.value === 'global') {
      const allChars = new Set<string>();
      batchSplitResult.segments.forEach((seg: any) => {
        (seg.characters || []).forEach((c: any) =>
          allChars.add(typeof c === 'string' ? c : c.name)
        );
      });
      setGlobalCharacterList(Array.from(allChars));
    }
  }}
  size="small"
>
  <Radio.Button value="individual">逐个绑定</Radio.Button>
  <Radio.Button value="global">全局绑定</Radio.Button>
</Radio.Group>
```
